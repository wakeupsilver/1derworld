// atlas.js
// One big texture holding every project's thumbnail, filled in as images arrive.
// All thumbnail surfaces (floor, disco ball, quilt, dream) sample this with per-instance UVs.
import * as THREE from 'three';
import { BUNDLE, canvas, drawCover, isPlaceholder, loadImage, pool, projectImage } from './util.js';
const CELL_W = 120;
const CELL_H = 90;
export class Atlas {
  constructor(projects) {
    this.projects = projects;
    const size = projects.length * CELL_W * CELL_H > 2048 * 2048 * 0.85 ? 4096 : 2048;
    [this.canvas, this.g] = canvas(size, size);
    this.size = size;
    this.cols = Math.floor(size / CELL_W);
    this.rows = Math.floor(size / CELL_H);
    this.images = new Array(projects.length).fill(null);   // real thumbnails only
    this.texture = new THREE.CanvasTexture(this.canvas);
    this.texture.colorSpace = THREE.SRGBColorSpace;
    this.texture.anisotropy = 8;
    this.dirty = false;
    this.lastFlush = 0;
    this.loaded = 0;
    this.listeners = [];
    // Draw covers straight away so nothing is ever blank while the real images load.
    // In the static edition real images arrive at once, so only the ones without an image need one.
    projects.forEach((p, i) => { if (!BUNDLE || !projectImage(p.id)) drawCover(this.g, ...this.cell(i), CELL_W
    this.texture.needsUpdate = true;
  }
  cell(i) {
    return [(i % this.cols) * CELL_W, Math.floor(i / this.cols) * CELL_H];
  }
  /** [u, v, width, height] of a thumbnail, inset to stop mipmap bleed between neighbours. */
  rect(i) {
    const [x, y] = this.cell(i);
    const inset = 2;
    return [
      (x + inset) / this.size,
      1 - (y + CELL_H - inset) / this.size,
      (CELL_W - inset * 2) / this.size,
      (CELL_H - inset * 2) / this.size,
    ];
  }
  onChange(fn) { this.listeners.push(fn); }
  async load(onProgress) {
    const order = [...this.projects].sort((a, b) => b.loves - a.loves);
    await pool(order, 6, async (p) => {
      const img = await loadImage(projectImage(p.id));
      if (img && !isPlaceholder(img)) {
        this.images[p.index] = img;
        const [x, y] = this.cell(p.index);
        this.g.drawImage(img, x, y, CELL_W, CELL_H);
        this.dirty = true;
      }
      this.loaded++;
      onProgress?.(this.loaded, this.projects.length);
    });
    this.dirty = true;
  }
  /** Called every frame. Re-uploading a big texture is costly, so it's throttled. */
  flush() {
    const now = performance.now() / 1000;
    if (!this.dirty || now - this.lastFlush < (this.size > 2048 ? 2 : 0.8)) return;
    this.dirty = false;
    this.lastFlush = now;
    this.texture.needsUpdate = true;
    for (const fn of this.listeners) fn();
  }
  /** Draw project i into any canvas, real thumbnail or generated cover. */
  draw(g, i, x, y, w, h) {
    const img = this.images[i];
    if (img) g.drawImage(img, x, y, w, h);
    else drawCover(g, x, y, w, h, this.projects[i]);
  }
  /** InstancedMesh whose instances each show one thumbnail. */
  instances(geometry, count, indexOf, materialOpts = {}) {
    const geo = geometry.clone();
    const auv = new Float32Array(count * 4);
    for (let k = 0; k < count; k++) auv.set(this.rect(indexOf(k)), k * 4);
    geo.setAttribute('auv', new THREE.InstancedBufferAttribute(auv, 4));
    const mat = new THREE.MeshBasicMaterial({ map: this.texture, toneMapped: false, ...materialOpts });
    mat.onBeforeCompile = (shader) => {
      shader.vertexShader = 'attribute vec4 auv;\n' + shader.vertexShader.replace(
        '#include <uv_vertex>',
        '#include <uv_vertex>\n  vMapUv = auv.xy + uv * auv.zw;',
      );
    };
    mat.customProgramCacheKey = () => 'atlas-thumb';
    return new THREE.InstancedMesh(geo, mat, count);
  }
}

// The globe, reimagined: a disco ball is already a sphere of little square tiles,
// so every facet is a project thumbnail. Hover one and it pops out. It also throws
// coloured light around the room.
import * as THREE from 'three';
import { canvas, canvasTexture, reducedMotion, seeded } from './util.js';
const ROWS = 15;
export function buildDisco(ctx, center, radius) {
  const { data, atlas, ui, scene } = ctx;
  const n = data.projects.length;
  const group = new THREE.Group();
  group.position.copy(center);
  scene.add(group);

  group.add(spinner);

  core.castShadow = true;

  const chain = new THREE.Mesh(new THREE.CylinderGeometry(0.018, 0.018, 1, 6), new THREE.MeshStandardMaterial(
  chain.raycast = () => {};
  scene.add(chain);
  // Rows of upright tiles from pole to pole, exactly how a mirror ball is built.
  const dummy = new THREE.Object3D();
  const placed = [];
  const rowHeight = (Math.PI * radius) / (ROWS + 1);
  for (let row = 0; row < ROWS; row++) {
    const lat = -Math.PI / 2 + ((row + 1) / (ROWS + 1)) * Math.PI;
    const ringR = Math.cos(lat) * radius;
    const count = Math.max(4, Math.round((2 * Math.PI * ringR) / (rowHeight * 4 / 3)));
    for (let i = 0; i < count; i++) {
      const a = ((i + (row % 2) * 0.5) / count) * Math.PI * 2;
      placed.push({
        dir: new THREE.Vector3(Math.cos(a) * Math.cos(lat), Math.sin(lat), Math.sin(a) * Math.cos(lat)),
        w: ((2 * Math.PI * ringR) / count) * 0.93,
        h: rowHeight * 0.93,
      });
    }
  }
  const FACETS = placed.length;
  // Shuffle which project sits where, so neighbours on the ball aren't all the same year.
  const rnd = seeded(data.seed);
  const order = data.projects.map((p) => p.index).sort(() => rnd() - 0.5);
  const facets = n ? atlas.instances(new THREE.PlaneGeometry(1, 1), FACETS, (k) => order[k % n]) : null;
  if (facets) {
    const setFacet = (k, scale) => {
      const f = placed[k];
      dummy.position.copy(f.dir).multiplyScalar(radius * (scale > 1 ? 1.12 : 1));
      dummy.lookAt(f.dir.clone().multiplyScalar(radius * 3));
      dummy.scale.set(f.w * scale, f.h * scale, 1);
      dummy.updateMatrix();
      facets.setMatrixAt(k, dummy.matrix);
    };
    placed.forEach((_, k) => setFacet(k, 1));
    spinner.add(facets);
    let hovered = -1;
    const setHovered = (k) => {
      if (k === hovered) return;
      if (hovered >= 0) setFacet(hovered, 1);
      hovered = k;
      if (k >= 0) setFacet(k, 2.2);
      facets.instanceMatrix.needsUpdate = true;
    };
    const pick = {
      hover(hit) {
        if (hit.instanceId === undefined || hit.object !== facets) { setHovered(-1); return { title: 'Disco ba
        setHovered(hit.instanceId);
        return { project: data.projects[order[hit.instanceId % n]] };
      },
      leave() { setHovered(-1); },
      click(hit) {
        if (hit.instanceId !== undefined && hit.object === facets) ui.openProject(data.projects[order[hit.inst
      },
    };
    group.userData.pick = pick;
    group.isHovered = () => hovered >= 0;
    facets.computeBoundingSphere();
  }
  ctx.pickables.push(group);
  // Light spots: an inside-out sphere of soft dots drawn over everything at low strength.
  const [c, g] = canvas(2048, 1024);
  const colors = ['#ff5a3c', '#ffc531', '#12bfa0', '#3aa0ff', '#ff6fb5', '#8a5cf6', '#ffffff'];
  for (let i = 0; i < 120; i++) {
    const x = rnd() * 2048, lat = Math.acos(rnd() * 2 - 1);
    const y = (lat / Math.PI) * 1024;
    const r = 6 + rnd() * 8;
    const grad = g.createRadialGradient(x, y, 0, x, y, r);
    const col = colors[Math.floor(rnd() * colors.length)];
    grad.addColorStop(0, col); grad.addColorStop(0.65, col); grad.addColorStop(1, col + '00');
    g.fillStyle = grad;
    // Stretch near the poles to cancel the equirectangular squeeze.
    g.save();
    g.translate(x, y);
    g.scale(1 / Math.max(0.25, Math.sin(lat)), 1);
    g.translate(-x, -y);
    g.beginPath(); g.arc(x, y, r, 0, 7); g.fill();
    g.restore();
  }
  const dots = new THREE.Mesh(new THREE.SphereGeometry(30, 32, 20), new THREE.MeshBasicMaterial({
    map: canvasTexture(c), side: THREE.BackSide, transparent: true, blending: THREE.AdditiveBlending,
    depthTest: false, depthWrite: false, opacity: 0.11, toneMapped: false,
  }));
  dots.renderOrder = 1;
  dots.raycast = () => {};
  group.add(dots);
  const spin = reducedMotion ? 0.03 : 0.16;
  const eyeRest = center.y;
  return {
    group,
    update(dt, t, camera) {
      if (!group.isHovered?.()) {
        spinner.rotation.y += spin * dt;
        dots.rotation.y -= spin * dt * 0.7;
      }
      spinner.rotation.z = Math.sin(t * 0.25) * 0.05;
      // The ball lowers to meet you when you walk up to it.
      const d = Math.hypot(camera.position.x - center.x, camera.position.z - center.z);
      const want = eyeRest - THREE.MathUtils.smoothstep(4.2 - d, 0, 2.2) * 0.85;
      group.position.y += (want - group.position.y) * (1 - Math.exp(-dt * 2));
      const len = ceilingY - group.position.y - radius;
      chain.scale.y = Math.max(0.01, len);
      chain.position.set(center.x, group.position.y + radius + len / 2, center.z);
    },
  };
}

import { assetURL } from './util.js';
const SCALE = [0, 2, 4, 7, 9];  // major pentatonic: any order of tiles sounds pleasant
export class Sound {
  constructor() {
    this.ctx = null;
    this.muted = false;
    this.dream = null;
  }
  /** Browsers only allow audio after a user gesture, so call this from one. */
  unlock() {
    if (!this.ctx) {
      const AC = window.AudioContext || window.webkitAudioContext;
      if (!AC) return;
      this.ctx = new AC();
      this.master = this.ctx.createGain();
      this.master.gain.value = this.muted ? 0 : 1;
      this.master.connect(this.ctx.destination);
    }
    if (this.ctx.state === 'suspended') this.ctx.resume();
  }
  setMuted(m) {
    this.muted = m;
    if (this.master) this.master.gain.setTargetAtTime(m ? 0 : 1, this.ctx.currentTime, 0.02);
  }
  tone(freq, { type = 'triangle', dur = 0.25, vol = 0.12, slide = 0, delay = 0 } = {}) {
    if (!this.ctx || this.muted) return;
    const t = this.ctx.currentTime + delay;
    const o = this.ctx.createOscillator();
    const g = this.ctx.createGain();
    o.type = type;
    o.frequency.setValueAtTime(freq, t);
    if (slide) o.frequency.exponentialRampToValueAtTime(Math.max(30, freq + slide), t + dur);
    g.gain.setValueAtTime(0.0001, t);
    g.gain.exponentialRampToValueAtTime(vol, t + 0.012);
    g.gain.exponentialRampToValueAtTime(0.0001, t + dur);
    o.connect(g).connect(this.master);
    o.start(t);
    o.stop(t + dur + 0.02);
  }

    const semis = SCALE[n % 5] + 12 * Math.floor(n / 5);
    this.tone(196 * Math.pow(2, semis / 12), { dur: 0.32, vol: 0.09 });
  }

  /** Soft pad plus the project's own sounds, drifting through a long echo. */
  startDream() {
    this.unlock();
    if (!this.ctx || this.dream) return;
    const ctx = this.ctx;
    const bus = ctx.createGain();
    bus.gain.value = 0;
    bus.gain.linearRampToValueAtTime(1, ctx.currentTime + 2.5);
    bus.connect(this.master);
    const seconds = 3.5;
    const impulse = ctx.createBuffer(2, ctx.sampleRate * seconds, ctx.sampleRate);
    for (let ch = 0; ch < 2; ch++) {
      const d = impulse.getChannelData(ch);
      for (let i = 0; i < d.length; i++) d[i] = (Math.random() * 2 - 1) * Math.pow(1 - i / d.length, 2.6);
    }
    const reverb = ctx.createConvolver();
    reverb.buffer = impulse;
    reverb.connect(bus);
    const filter = ctx.createBiquadFilter();
    filter.type = 'lowpass';
    filter.frequency.value = 600;
    filter.connect(reverb);
    const padGain = ctx.createGain();
    padGain.gain.value = 0.05;
    padGain.connect(filter);
    const oscillators = [];
    [0, 7, 12, 16].forEach((semi, i) => {
      for (const detune of [-6, 6]) {
        const o = ctx.createOscillator();
        o.type = i % 2 ? 'triangle' : 'sine';
        o.frequency.value = 110 * Math.pow(2, semi / 12);
        o.detune.value = detune;
        o.connect(padGain);
        o.start();
        oscillators.push({ o, semi });
      }
    });
    const chords = [0, -3, 5, 2];
    let chord = 0;
    const chordTimer = setInterval(() => {
      chord = (chord + 1) % chords.length;
      for (const { o, semi } of oscillators) {
        o.frequency.setTargetAtTime(110 * Math.pow(2, (semi + chords[chord]) / 12), ctx.currentTime, 1.4);
      }
      filter.frequency.setTargetAtTime(400 + Math.random() * 700, ctx.currentTime, 2);
    }, 7000);
    this.dreamBuffers ||= [];
    const state = { bus, oscillators, chordTimer, sampleTimer: null, stopped: false };
    this.dream = state;
    const rates = [0.5, 0.75, 1, 1.125, 1.5];
    state.sampleTimer = setInterval(() => {
      if (!this.dreamBuffers.length || state.stopped) return;
      const src = ctx.createBufferSource();
      src.buffer = this.dreamBuffers[Math.floor(Math.random() * this.dreamBuffers.length)];
      src.playbackRate.value = rates[Math.floor(Math.random() * rates.length)];
      const g = ctx.createGain();
      g.gain.value = 0.32;
      const pan = ctx.createStereoPanner ? ctx.createStereoPanner() : null;
      if (pan) { pan.pan.value = Math.random() * 1.6 - 0.8; src.connect(g).connect(pan).connect(reverb); } els
      src.start();
      src.stop(ctx.currentTime + 8);
    }, 3200);
  }
  /** Decode a project's sounds into the running dream. Fetched once, then kept for later naps. */
  async addDreamSounds(sounds) {
    this.dreamBuffers ||= [];
    // Old Scratch sounds use a compressed WAV type browsers can't decode, so skip those up front.
    const usable = sounds.filter((s) => !s.adpcm && s.secs > 0 && s.secs < 25).slice(0, 10);
    await Promise.all(usable.map(async (s) => {
      try {
        const buf = await (await fetch(assetURL(s.md5))).arrayBuffer();
        this.dreamBuffers.push(await this.ctx.decodeAudioData(buf));
      } catch { /* a sound that won't decode just doesn't join the mix */ }
    }));
  }
  stopDream() {
    const d = this.dream;
    if (!d) return;
    this.dream = null;
    d.stopped = true;
    clearInterval(d.chordTimer);
    clearInterval(d.sampleTimer);
    const ctx = this.ctx;
    d.bus.gain.cancelScheduledValues(ctx.currentTime);
    d.bus.gain.setTargetAtTime(0, ctx.currentTime, 0.4);
    setTimeout(() => {
      for (const { o } of d.oscillators) { try { o.stop(); } catch { /* already stopped */ } }
      d.bus.disconnect();
    }, 2000);
  }
}

          actor.position.y = actor.home.y + Math.abs(Math.sin(p * 1.4)) * 0.07;
        }
      }
    },
  };

    vec2 i = floor(p), f = fract(p); f = f*f*(3.0-2.0*f);
    return mix(mix(hash(i), hash(i+vec2(1,0)), f.x), mix(hash(i+vec2(0,1)), hash(i+vec2(1,1)), f.x), f.y);

  vec3 sampleCell(vec4 r, vec2 uv){

  }`;

export class Rig {
  constructor(camera, opts = {}) {
    this.camera = camera;
    camera.rotation.order = 'YXZ';
    this.yaw = 0;
    this.pitch = 0;
    this.eye = opts.eye ?? 1.55;
    this.fly = !!opts.fly;
    this.speed = opts.speed ?? 3.6;
    this.limit = opts.limit ?? ((p) => p);
    this.vel = new THREE.Vector3();
    this.glideState = null;
    this.locked = false;
  }
  place({ x, y, z, yaw = 0, pitch = 0 }) {
    this.camera.position.set(x, y, z);
    this.yaw = yaw; this.pitch = pitch;
    this.apply();
  }
  apply() {
    this.camera.rotation.set(this.pitch, this.yaw, 0);
  }
  /** Glide smoothly to a pose. Takes the short way round when turning. */
  glide(to, dur = 1.1, done) {
    const from = { x: this.camera.position.x, y: this.camera.position.y, z: this.camera.position.z, yaw: this.
    let dy = (to.yaw ?? this.yaw) - this.yaw;
    dy = Math.atan2(Math.sin(dy), Math.cos(dy));
    this.glideState = { from, to: { ...from, ...to, yaw: this.yaw + dy }, t: 0, dur, done };
  }
  update(dt, move) {
    const moving = move.x !== 0 || move.z !== 0 || move.y !== 0;
    if (this.glideState && moving && !this.locked) this.glideState = null;
    if (this.glideState) {
      const g = this.glideState;
      g.t = Math.min(1, g.t + dt / g.dur);
      const k = ease(g.t);
      const lerp = (a, b) => a + (b - a) * k;
      this.camera.position.set(lerp(g.from.x, g.to.x), lerp(g.from.y, g.to.y), lerp(g.from.z, g.to.z));
      this.yaw = lerp(g.from.yaw, g.to.yaw);
      this.pitch = lerp(g.from.pitch, g.to.pitch);
      this.apply();
      if (g.t >= 1) { this.glideState = null; g.done && g.done(); }
      return;
    }
    if (this.locked) return;
    const sin = Math.sin(this.yaw), cos = Math.cos(this.yaw);
    const speed = this.speed * (move.run ? 1.8 : 1);
    const want = new THREE.Vector3(
      (move.x * cos + move.z * sin) * speed,
      0,
      (-move.x * sin + move.z * cos) * speed,
    );
    if (this.fly) {
      const cp = Math.cos(this.pitch);
      want.set((move.x * cos - move.z * -sin * cp) * speed, (-move.z * Math.sin(this.pitch) + move.y) * speed,
    }
    this.vel.lerp(want, 1 - Math.exp(-dt * 12));
    const p = this.camera.position;
    p.addScaledVector(this.vel, dt);
    if (!this.fly && this.vel.lengthSq() > 0.01) p.y += (this.eye - p.y) * (1 - Math.exp(-dt * 4));
    this.limit(p);
    this.apply();
  }
}

      e.preventDefault();

/** Yaw and pitch that make a camera at `from` look at `to`. */
export function lookAngles(from, to) {
  const dx = to[0] - from[0], dy = to[1] - from[1], dz = to[2] - from[2];
  return { yaw: Math.atan2(-dx, -dz), pitch: Math.atan2(dy, Math.hypot(dx, dz)) };
}
/** One clean pose object for a station. */
export function pose(pos, target) {
  return { x: pos[0], y: pos[1], z: pos[2], ...lookAngles(pos, target) };
}

/** Largest tile size that still leaves a free cell for every project (furniture covers some). */
function layout(n, room, blocked) {
  const goal = Math.ceil(n * 1.04);
  let size = 1.3;
  for (;;) {
    const cols = Math.floor(room.width / size);
    const rows = Math.floor(room.depth / size);
    const stepX = room.width / cols;
    const stepZ = room.depth / rows;
    const free = [];
    // Row 0 is by the door, where you start, so walking forward is walking forward in time.
    for (let r = 0; r < rows; r++) {
      const line = [];
      for (let c = 0; c < cols; c++) {
        const x = -room.width / 2 + stepX * (c + 0.5);
        const z = room.depth / 2 - stepZ * (r + 0.5);
        const covered = blocked.some((b) => x > b.x0 && x < b.x1 && z > b.z0 && z < b.z1);
        line.push({ c, r, x, z, covered });
      }
      if (r % 2) line.reverse();
      free.push(...line);
    }
    if (free.filter((t) => !t.covered).length >= goal || size < 0.1) return { cells: free, stepX, stepZ, cols,
    size *= 0.97;
  }
}

  if (rings) { rings.raycast = () => {}; rings.frustumCulled = false; scene.add(rings); }
  // A little flag on the first tile of each year.
  const flags = [];
  let flagYear = null;
  const tileMin = Math.min(stepX, stepZ);
  for (const cell of withProject) {
    if (cell.project.year !== flagYear) {
      flagYear = cell.project.year;
      const s = labelSprite(String(flagYear), { bg: '#' + yearColor(flagYear, firstYear, lastYear).multiplySca
      s.position.set(cell.x, 0.26, cell.z);
      flags.push({ s, cell });
      scene.add(s);
    }
  }
  // Heights animate between 0 and each tile's tower height (see setHeights).
  cells.forEach((cell) => { cell.h = 0; cell.target = 0; });
  const m = new THREE.Matrix4();
  const q = new THREE.Quaternion();
  const up = new THREE.Vector3(0, 1, 0);
  const pos = new THREE.Vector3();
  const scl = new THREE.Vector3();
  const writeTile = (i) => {
    const c = cells[i];
    const press = c.press || 0;
    const top = c.h - press;
    const height = TILE_THICK + c.h;
    pos.set(c.x, top - height / 2, c.z);
    scl.set(stepX * gap, height / TILE_THICK, stepZ * gap);
    m.compose(pos, q.identity(), scl);
    tiles.setMatrixAt(i, m);
  };
  const writeThumb = (k) => {
    const c = withProject[k];
    pos.set(c.x, c.h - (c.press || 0) + 0.003, c.z);
    scl.set(stepX * gap * THUMB_FRACTION, 1, stepZ * gap * THUMB_FRACTION);
    m.compose(pos, q.identity(), scl);
    thumbs.setMatrixAt(k, m);
  };
  const writeRing = (k) => {
    const c = ringCells[k];
    const r = tileMin * 0.05;
    pos.set(c.x - stepX * 0.385, c.h + r * 0.3, c.z - stepZ * 0.385);
    scl.setScalar(r);
    m.compose(pos, q.identity(), scl);
    rings.setMatrixAt(k, m);
  };
  const writeGem = (k, t) => {
    const c = gemCells[k];
    const r = tileMin * (c.project.tier === 3 ? 0.13 : 0.085);
    pos.set(c.x + stepX * 0.37, c.h + r * 1.5 + Math.sin(t * 2 + k) * r * 0.25, c.z - stepZ * 0.37);
    q.setFromAxisAngle(up, t * 1.2 + k);
    scl.set(r, r * 1.35, r);
    m.compose(pos, q, scl);
    gems.setMatrixAt(k, m);
    q.identity();
  };
  const writeAll = () => {
    cells.forEach((_, i) => writeTile(i));
    withProject.forEach((_, k) => writeThumb(k));
    ringCells.forEach((_, k) => writeRing(k));
    tiles.instanceMatrix.needsUpdate = true;
    thumbs.instanceMatrix.needsUpdate = true;
    if (rings) rings.instanceMatrix.needsUpdate = true;
  };
  writeAll();
  const cellAt = (x, z) => {
    const c = Math.floor((x + room.width / 2) / stepX);
    const r = Math.floor((room.depth / 2 - z) / stepZ);
    return c >= 0 && c < cols && r >= 0 && r < rows ? grid[r * cols + c] : null;
  };
  const indexOfCell = new Map(cells.map((c, i) => [c, i]));
  let hover = null;
  const tint = (cell, amount) => {
    const i = indexOfCell.get(cell);
    tiles.setColorAt(i, color.copy(baseColors[i]).lerp(new THREE.Color('#ffffff'), amount));
    tiles.instanceColor.needsUpdate = true;
  };
  const setHoverCell = (cell) => {
    if (hover === cell) return;
    if (hover) tint(hover, 0);
    hover = cell;
    if (hover) tint(hover, 0.55);
  };
  // Invisible plane: cheaper to pick against than 1000+ instanced boxes.
  const proxy = new THREE.Mesh(new THREE.PlaneGeometry(room.width, room.depth).rotateX(-Math.PI / 2), new THRE
  proxy.visible = false;
  proxy.userData.pick = {
    hover(hit) {
      const cell = cellAt(hit.point.x, hit.point.z);
      setHoverCell(cell?.project ? cell : null);
      return cell?.project ? { project: cell.project } : null;
    },
    leave() { setHoverCell(null); },
    click(hit) {
      const cell = cellAt(hit.point.x, hit.point.z);
      if (cell?.project) ui.openProject(cell.project);
    },
  };
  scene.add(proxy);
  let underfoot = null;
  let animating = false;
  const pressed = new Set();
  return {
    proxy,
    cellAt,
    cells,
    /** Raise tiles by value (0..1 of the tallest), or flatten with null. */
    setHeights(valueOf) {
      for (const c of cells) c.target = c.project && valueOf ? valueOf(c.project) * 1.35 : 0;
      animating = true;
    },
    update(dt, t, camera) {
      if (gems) {
        gemCells.forEach((_, k) => writeGem(k, t));
        gems.instanceMatrix.needsUpdate = true;
      }
      const cell = camera.position.y < 2.2 ? cellAt(camera.position.x, camera.position.z) : null;
      if (cell !== underfoot) {
        underfoot = cell;
        if (cell?.project) {
          sound.step(cell.project.index / Math.max(1, projects.length - 1));
          cell.press = 0.035;
          pressed.add(cell);
          tint(cell, 0.4);
          ctx.onStep?.(cell.project);
        }
      }
      for (const c of pressed) {
        if (c === underfoot) continue;
        c.press = Math.max(0, c.press - dt * 0.12);
        if (c.press === 0) { pressed.delete(c); if (c !== hover) tint(c, 0); }
      }
      let changed = pressed.size > 0;
      if (animating) {
        let still = false;
        for (const c of cells) {
          const d = c.target - c.h;
          if (Math.abs(d) > 0.002) { c.h += d * (1 - Math.exp(-dt * 5)); still = true; } else c.h = c.target;
        }
        animating = still;
        changed = true;
        for (const f of flags) f.s.position.y = f.cell.h + 0.26;
      }
      if (changed) writeAll();
    },
  };
}
