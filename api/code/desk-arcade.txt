/* ---------------- Color math ---------------- */
let _h = 0, _s = 0, _v = 0;
function hsvFast(r, g, b) {
  const mx = Math.max(r, g, b), mn = Math.min(r, g, b), d = mx - mn;
  _v = mx / 255; _s = mx ? d / mx : 0;
  if (!d) _h = 0;
  else if (mx === r) _h = 60 * (((g - b) / d) % 6);
  else if (mx === g) _h = 60 * ((b - r) / d + 2);
  else _h = 60 * ((r - g) / d + 4);
  if (_h < 0) _h += 360;
}
function hsv(r, g, b) { hsvFast(r, g, b); return [_h, _s, _v]; }
function cdist(h, s, v, ru) {
  let dh = Math.abs(h - ru.h); if (dh > 180) dh = 360 - dh; dh /= 180;
  const bright = Math.max(v, ru.v);                 // saturation/hue are unreliable in the dark
  const a = dh * Math.min(s, ru.s) * 2.2 * Math.min(1, bright * 1.4);
  const b = (s - ru.s) * bright;
  const c = (v - ru.v) * 0.85;
  return Math.sqrt(a * a + b * b + c * c);
}

function classify() {
  sctx.imageSmoothingEnabled = true; sctx.imageSmoothingQuality = 'high';
  sctx.drawImage(fc, 0, 0, GW, GH);
  const d = sctx.getImageData(0, 0, GW, GH).data;
  const n = GW * GH, rl = rules.length;
  const codes = rules.map(r => MEANINGS[r.meaning].code);
  for (let i = 0; i < n; i++) {
    const p = i * 4;
    hsvFast(d[p], d[p + 1], d[p + 2]);
    let best = 0, bd = 1e9;
    for (let k = 0; k < rl; k++) {
      const ru = rules[k], dd = cdist(_h, _s, _v, ru);
      if (dd < ru.tol && dd < bd) { bd = dd; best = codes[k]; }
    }
    tmp[i] = best;
  }
  // remove speckle noise: a cell needs at least 2 matching neighbours
  for (let y = 0; y < GH; y++) for (let x = 0; x < GW; x++) {
    const i = y * GW + x, c = tmp[i];
    if (!c) { grid[i] = 0; continue; }
    let same = 0;
    for (let dy = -1; dy <= 1; dy++) for (let dx = -1; dx <= 1; dx++) {
      if (!dx && !dy) continue;
      const xx = x + dx, yy = y + dy;
      if (xx >= 0 && yy >= 0 && xx < GW && yy < GH && tmp[yy * GW + xx] === c) same++;
