// Tiny collision world: boxes (floors you stand on, walls you bump), height functions (ramps, halfpipes, hill
// sample(x,z,y) -> the highest walkable surface at or a little above y.  blocked(x,z,y,r) -> is there wall he
export const STEP = 0.55;
export class Terrain {
  constructor({ base, ceiling = () => 999, extraBlock } = {}) {
    this.boxes = [];
    this.fns = [];
    this.base = base || (() => ({ h: 0, kind: 'floor' }));
    this.ceilingFn = ceiling;
    this.extraBlock = extraBlock;
  }
  box(minX, maxX, minZ, maxZ, top, o = {}) {
    const b = { minX, maxX, minZ, maxZ, top, bottom: o.bottom ?? 0, kind: o.kind || 'floor', data: o.data, act
    this.boxes.push(b);
    return b;
  }
  // wall: a box you cannot stand on (top far above)
  wall(minX, maxX, minZ, maxZ, top = 40, o = {}) { return this.box(minX, maxX, minZ, maxZ, top, { ...o, wallOn
  fn(minX, maxX, minZ, maxZ, h, o = {}) {
    const f = { minX, maxX, minZ, maxZ, h, kind: o.kind || 'floor', data: o.data, active: o.active ?? true };
    this.fns.push(f);
    return f;
  }
  sample(x, z, y) {
    let best = this.base(x, z, y);
    if (best.h > y + STEP) best = { h: -999, kind: 'void' };
    for (const b of this.boxes) {
      if (!b.active || b.wallOnly || x < b.minX || x > b.maxX || z < b.minZ || z > b.maxZ) continue;
      if (b.top <= y + STEP && b.top > best.h) best = { h: b.top, kind: b.kind, data: b.data };
    }
    for (const f of this.fns) {
      if (!f.active || x < f.minX || x > f.maxX || z < f.minZ || z > f.maxZ) continue;
      const h = f.h(x, z);
      if (h <= y + STEP && h >= best.h) best = { h, kind: f.kind, data: f.data }; // >= so a ramp's flat part 
    }
    return best;
  }
  blocked(x, z, y, r = 0.3) {
    if (this.extraBlock && this.extraBlock(x, z, y)) return true;
    for (const b of this.boxes) {
      if (!b.active) continue;
      if (x < b.minX - r || x > b.maxX + r || z < b.minZ - r || z > b.maxZ + r) continue;
      if (b.top > y + STEP && b.bottom < y + 1.75) return true;
    }
    for (const f of this.fns) {
      if (!f.active || x < f.minX || x > f.maxX || z < f.minZ || z > f.maxZ) continue;
      if (f.h(x, z) > y + STEP) return true;
    }
    return false;
  }
  ceiling(x, z, y = 0) {
    let c = this.ceilingFn(x, z, y);
    for (const b of this.boxes) {
      if (!b.active || x < b.minX || x > b.maxX || z < b.minZ || z > b.maxZ) continue;
      if (b.bottom > y + 1 && b.bottom < c) c = b.bottom;
    }
    return c;
  }
}

// ---- Code atlas: real source code drawn as lines for the code-stream shader ----
export function codeAtlas(text, w = 1024, h = 2048, fontPx = 22) {
  const c = canvas(w, h);
  const g = c.getContext('2d');
  g.fillStyle = '#000'; g.fillRect(0, 0, w, h);
  g.font = `500 ${fontPx}px "JetBrains Mono", monospace`;
  g.textBaseline = 'top';
  const lineH = fontPx * 1.25;
  const lines = text.split('\n').filter(l => l.trim());
  const rows = Math.floor(h / lineH);
  const cols = Math.floor(w / (fontPx * 0.6));
  for (let r = 0, i = 0; r < rows; r++, i++) {
    const l = (lines[i % Math.max(1, lines.length)] || '').replace(/\t/g, '  ');
    const indent = l.length - l.trimStart().length;
    const isComment = /^\s*(\/\/|#)/.test(l);
    g.fillStyle = isComment ? '#ff0000' : '#ffffff'; // red channel marks comments: file names show as bright 
    g.fillText(l.slice(0, cols), 6 + Math.min(indent, 12) * 2, r * lineH + 2);
  }
  const t = tex(c, { srgb: false });
  t.wrapS = t.wrapT = THREE.RepeatWrapping;
  return t;
}

    this.vel.set(d.x / T, (d.y - 0.5 * G * T * T) / T, d.z / T);

    const f = (k.KeyW || k.ArrowUp ? 1 : 0) - (k.KeyS || k.ArrowDown ? 1 : 0) - tm.y;
    const r = (k.KeyD ? 1 : 0) - (k.KeyA ? 1 : 0) + tm.x;
    // arrows turn (Makey Makey friendly)
    if (this.enabled) {
      this.yaw += ((k.ArrowLeft ? 1 : 0) - (k.ArrowRight ? 1 : 0)) * 2.3 * dt;
    }
    const run = k.ShiftLeft || k.ShiftRight || Math.hypot(tm.x, tm.y) > 0.9;
    const fwd = this.forward(), right = new THREE.Vector3(-fwd.z, 0, fwd.x);
    const wish = fwd.multiplyScalar(f).add(right.multiplyScalar(r));
    if (wish.lengthSq() > 1) wish.normalize();
    if (!this.enabled) wish.set(0, 0, 0);

    const boost = this.boost > 0 ? 1.4 : 1;   // taco power

    const accel = this.grounded ? (skate ? 4 : 11) : 1.8;

    // vertical

      } else {
